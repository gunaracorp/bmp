   178→export async function exportCmsDatabaseSnapshot(): Promise<CmsDatabaseBackupSnapshot> {
   179→  const client = rawCmsClient;
   180→  const capturedAt = new Date().toISOString();
   181→  let transactionOpen = false;
   182→
   183→  try {
   184→    await client.query('BEGIN ISOLATION LEVEL REPEATABLE READ READ ONLY');
   185→    transactionOpen = true;
   186→
   187→    const { rows: liveTables } = await client.query<{ table_name: string }>(
   188→      'SELECT table_name FROM information_schema.tables WHERE table_schema = $1 AND table_type = $2 ORDER BY table_name',
   189→      ['public', 'BASE TABLE'],
   190→    );
   191→    const liveTableNames = new Set(liveTables.map(({ table_name }) => table_name));
   192→    const expectedTableNames = new Set<string>(CMS_BACKUP_PUBLIC_TABLES);
   193→    const missingTables = CMS_BACKUP_PUBLIC_TABLES.filter((tableName) => !liveTableNames.has(tableName));
   194→    const unexpectedTables = liveTables.map(({ table_name }) => table_name).filter((tableName) => !expectedTableNames.has(tableName));
   195→
   196→    if (missingTables.length || unexpectedTables.length) {
   197→      throw new Error(`CMS_BACKUP_SCHEMA_MISMATCH: missing=${missingTables.join(',')} unexpected=${unexpectedTables.join(',')}`);
   198→    }
   199→
   200→    const tables: CmsDatabaseBackupSnapshot['tables'] = [];
   201→
   202→    for (const tableName of CMS_BACKUP_PUBLIC_TABLES) {
   203→      const [{ rows: columnRows }, { rows: keyRows }, { rows }] = await Promise.all([
   204→        client.query<{
   205→          column_name: string;
   206→          data_type: string;
   207→          is_nullable: 'YES' | 'NO';
   208→          column_default: string | null;
   209→        }>(
   210→          'SELECT column_name, data_type, is_nullable, column_default FROM information_schema.columns WHERE table_schema = $1 AND table_name = $2 ORDER BY ordinal_position',
   211→          ['public', tableName],
   212→        ),
   213→        client.query<{ column_name: string }>(
   214→          'SELECT key_column_usage.column_name FROM information_schema.table_constraints JOIN information_schema.key_column_usage USING (constraint_catalog, constraint_schema, constraint_name, table_name) WHERE table_constraints.table_schema = $1 AND table_constraints.table_name = $2 AND table_constraints.constraint_type = $3 ORDER BY key_column_usage.ordinal_position',
   215→          ['public', tableName, 'PRIMARY KEY'],
   216→        ),
   217→        client.query<Record<string, unknown>>(`SELECT * FROM "public".${quotePostgresIdentifier(tableName)}`),
   218→      ]);
   219→
   220→      tables.push({
   221→        name: tableName,
   222→        columns: columnRows.map((column) => ({
   223→          name: column.column_name,
   224→          type: column.data_type,
   225→          nullable: column.is_nullable === 'YES',
   226→          default: column.column_default,
   227→        })),
   228→        primaryKey: keyRows.map(({ column_name }) => column_name),
   229→        rowCount: rows.length,
   230→        rows,
   231→      });
   232→    }
   233→
   234→    await client.query('COMMIT');
   235→    transactionOpen = false;
   236→
   237→    return {
   238→      format: 'berkat-mandiri-cms-backup',
   239→      version: 1,
   240→      capturedAt,
   241→      database: 'postgres',
   242→      schema: 'public',
   243→      tables,
   244→    };
   245→  } catch (error) {
   246→    if (transactionOpen) await client.query('ROLLBACK').catch(() => undefined);
   247→    throw error;
   248→  } finally {
   249→    client.release();
   250→  }
   251→}

   303→export async function getProductImageAssetStats(ownerUserId: string) {
   304→  const approvedStatuses = ['OWNED', 'LICENSED', 'SUPPLIER_AUTHORIZED', 'MANUFACTURER_AUTHORIZED', 'PARTNER_AUTHORIZED', 'PUBLIC_REUSE_PERMITTED'];
   305→  const [rightsRows, processingRows, actionableRows] = await Promise.all([
   306→    cmsDb.select({ status: productImageAssets.rightsStatus, count: sql<number>`count(*)::int` })
   307→      .from(productImageAssets)
   308→      .where(and(eq(productImageAssets.ownerUserId, ownerUserId), isNull(productImageAssets.sourceDeletedAt)))
   309→      .groupBy(productImageAssets.rightsStatus),
   310→    cmsDb.select({ status: productImageAssets.processingStatus, count: sql<number>`count(*)::int` })
   311→      .from(productImageAssets)
   312→      .where(and(eq(productImageAssets.ownerUserId, ownerUserId), isNull(productImageAssets.sourceDeletedAt)))
   313→      .groupBy(productImageAssets.processingStatus),
   314→    cmsDb.select({ status: productImageAssets.processingStatus, count: sql<number>`count(*)::int` })
   315→      .from(productImageAssets)
   316→      .where(and(
   317→        eq(productImageAssets.ownerUserId, ownerUserId),
   318→        isNull(productImageAssets.sourceDeletedAt),
   319→        inArray(productImageAssets.rightsStatus, approvedStatuses),
   320→        inArray(productImageAssets.processingStatus, ['DISCOVERED', 'FAILED']),
   321→      ))
   322→      .groupBy(productImageAssets.processingStatus),
   323→  ]);
   324→  const rights = Object.fromEntries(rightsRows.map(({ status, count }) => [status, count]));
   325→  const processing = Object.fromEntries(processingRows.map(({ status, count }) => [status, count]));
   326→  const actionable = Object.fromEntries(actionableRows.map(({ status, count }) => [status, count]));
   327→  return {
   328→    total: Object.values(rights).reduce((sum, count) => sum + count, 0),
   329→    pending: rights.PENDING_REVIEW ?? 0,
   330→    approved: ['OWNED', 'LICENSED', 'SUPPLIER_AUTHORIZED', 'MANUFACTURER_AUTHORIZED', 'PARTNER_AUTHORIZED', 'PUBLIC_REUSE_PERMITTED'].reduce((sum, status) => sum + (rights[status] ?? 0), 0),
   331→    rejected: rights.REJECTED ?? 0,
   332→    ready: actionable.DISCOVERED ?? 0,
   333→    retryable: actionable.FAILED ?? 0,
   334→    processed: processing.PROCESSED ?? 0,
   335→    failed: processing.FAILED ?? 0,
   336→    invalid: processing.INVALID ?? 0,
   337→    duplicates: processing.DUPLICATE ?? 0,
   338→    processing: processing.PROCESSING ?? 0,
   339→  };
   340→}

   342→export async function claimProductImageAssets(ownerUserId: string, limit = 10, retryFailed = false) {
   343→  const staleBefore = new Date(Date.now() - 20 * 60 * 1000);
   344→  const claimableStatuses = retryFailed ? ['DISCOVERED', 'FAILED'] : ['DISCOVERED'];
   345→  return cmsDb.transaction(async (transaction) => {
   346→    const assets = await transaction.select().from(productImageAssets).where(and(
   347→      eq(productImageAssets.ownerUserId, ownerUserId),
   348→      isNull(productImageAssets.sourceDeletedAt),
   349→      inArray(productImageAssets.rightsStatus, ['OWNED', 'LICENSED', 'SUPPLIER_AUTHORIZED', 'MANUFACTURER_AUTHORIZED', 'PARTNER_AUTHORIZED', 'PUBLIC_REUSE_PERMITTED']),
   350→      or(
   351→        inArray(productImageAssets.processingStatus, claimableStatuses),
   352→        and(eq(productImageAssets.processingStatus, 'PROCESSING'), lt(productImageAssets.updatedAt, staleBefore)),
   353→      ),
   354→    )).orderBy(productImageAssets.createdAt).limit(Math.min(Math.max(limit, 1), 10)).for('update', { skipLocked: true });
   355→
   356→    if (assets.length) {
   357→      await transaction.update(productImageAssets).set({ processingStatus: 'PROCESSING', processingError: null, updatedAt: new Date() })
   358→        .where(and(eq(productImageAssets.ownerUserId, ownerUserId), inArray(productImageAssets.id, assets.map((asset) => asset.id))));
   359→    }
   360→    return assets;
   361→  });
   362→}

   751→export async function getCatalogProductsPage(
   752→  query: string,
   753→  categorySlug: string,
   754→  pageSize: number,
   755→  offset: number,
   756→) {
   757→  const { and, count, eq, ilike, or } = await import('drizzle-orm');
   758→  const predicates = [eq(products.inStock, true)];
   759→  const normalizedQuery = query.trim().slice(0, 100);
   760→  if (normalizedQuery) {
   761→    const escapedQuery = normalizedQuery.replace(/[\\%_]/g, '\\$&');
   762→    const search = `%${escapedQuery}%`;
   763→    predicates.push(or(
   764→      ilike(products.name, search),
   765→      ilike(products.model, search),
   766→      ilike(products.brand, search),
   767→      ilike(products.shortDesc, search),
   768→    )!);
   769→  }
   770→  if (categorySlug) {
   771→    predicates.push(eq(categories.slug, categorySlug));
   772→  }
   773→
   774→  const where = and(...predicates);
   775→  const [rows, [{ total }]] = await Promise.all([
   776→    cmsDb.select({ product: products, category: { name: categories.name, slug: categories.slug } })
   777→      .from(products)
   778→      .innerJoin(categories, eq(products.categoryId, categories.id))
   779→      .where(where)
   780→      .orderBy(products.name)
   781→      .limit(Math.min(Math.max(pageSize, 1), 48))
   782→      .offset(Math.max(offset, 0)),
   783→    cmsDb.select({ total: count() })
   784→      .from(products)
   785→      .innerJoin(categories, eq(products.categoryId, categories.id))
   786→      .where(where),
   787→  ]);
   788→
   789→  return { products: rows, total: Number(total) };
   790→}
